-- Crear base de datos (opcional, para pruebas locales)
CREATE DATABASE IF NOT EXISTS fitzone_api;
USE fitzone_api;

-- 1 Creacion de tabla
CREATE TABLE socios (
    id_socio       INT PRIMARY KEY,
    nombres        VARCHAR(50)  NOT NULL,
    apellidos      VARCHAR(50)  NOT NULL,
    edad           INT          NOT NULL,
    fecha_registro VARCHAR(10)  NOT NULL,  -- Formato YYYY-MM-DD
    activo         BOOLEAN      NOT NULL
);

--3 Consultas

-- Consulta 1: nombres, apellidos y edad de socios activos mayores de 18 años
SELECT nombres, apellidos, edad
FROM socios
WHERE activo = TRUE AND edad > 18;

-- Consulta 2: socios registrados después de '2026-02-01'
SELECT *
FROM socios
WHERE fecha_registro > '2026-02-01';

-- Consulta 3: todos los socios ordenados por edad de mayor a menor
SELECT *
FROM socios
ORDER BY edad DESC;
